# Vello 2.0 Frontend Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the complete Vello 2.0 web frontend — a platform where anyone can launch, fund, and earn from autonomous AI Influencers — using Next.js 15 (App Router) + shadcn/ui.

**Architecture:** Next.js 15 App Router with TypeScript, shadcn/ui as the component system, and Tailwind CSS for styling. The app is organized around four surfaces (Studio, Market, Stage, Wallet) each as a top-level route segment. Data is mocked via static fixtures in this phase; API integration is a separate plan.

**Tech Stack:** Next.js 15, TypeScript, shadcn/ui, Tailwind CSS v4, Lucide Icons, Zustand (client state), React Hook Form + Zod (forms), Recharts (analytics charts)

> **⚠️ Scope Note:** Vello 2.0 is three additional independent subsystems:
> - `2026-05-26-vello-2-solana-contracts.md` — Anchor smart contracts (Registry, Vault, Share Issuer, Escrow)
> - `2026-05-26-vello-2-ai-pipeline.md` — FLUX LoRA, video gen, voice clone pipeline
> - `2026-05-26-vello-2-backend.md` — Node.js API, Postgres, job queues
>
> This plan delivers the frontend shell only. Backend integration is wired via mock data; swap for real API calls when the backend plan completes.

```typescript
import { create } from "zustand";

export type WizardStep =
  | "identity"
  | "voice"
  | "personality"
  | "brand"
  | "behavior"
  | "launch";

export const WIZARD_STEPS: WizardStep[] = [
  "identity",
  "voice",
  "personality",
  "brand",
  "behavior",
  "launch",
];

export interface WizardState {
  currentStep: WizardStep;
  identity: {
    name: string;
    handle: string;
    referencePhotoUrl: string;
    archetype: string;
  };
  voice: {
    voiceId: string;
    language: string;
    tone: string;
  };
  personality: {
    warmth: number;
    humor: number;
    authority: number;
    boldness: number;
    niche: string;
  };
  brand: {
    description: string;
    categories: string[];
    isHalal: boolean;
    bannedCategories: string[];
  };
  behavior: {
    autoAcceptBelow: number;
    autoDeclineCategories: string[];
    operationalPoolPercent: number;
  };
  goToStep: (step: WizardStep) => void;
  nextStep: () => void;
  prevStep: () => void;
  updateIdentity: (data: Partial<WizardState["identity"]>) => void;
  updateVoice: (data: Partial<WizardState["voice"]>) => void;
  updatePersonality: (data: Partial<WizardState["personality"]>) => void;
  updateBrand: (data: Partial<WizardState["brand"]>) => void;
  updateBehavior: (data: Partial<WizardState["behavior"]>) => void;
}

export const useWizardStore = create<WizardState>((set, get) => ({
  currentStep: "identity",
  identity: { name: "", handle: "", referencePhotoUrl: "", archetype: "" },
  voice: { voiceId: "", language: "en", tone: "warm" },
  personality: { warmth: 70, humor: 50, authority: 50, boldness: 50, niche: "lifestyle" },
  brand: { description: "", categories: [], isHalal: false, bannedCategories: [] },
  behavior: { autoAcceptBelow: 1000, autoDeclineCategories: [], operationalPoolPercent: 15 },

  goToStep: (step) => set({ currentStep: step }),

  nextStep: () => {
    const idx = WIZARD_STEPS.indexOf(get().currentStep);
    if (idx < WIZARD_STEPS.length - 1) {
      set({ currentStep: WIZARD_STEPS[idx + 1] });
    }
  },

  prevStep: () => {
    const idx = WIZARD_STEPS.indexOf(get().currentStep);
    if (idx > 0) {
      set({ currentStep: WIZARD_STEPS[idx - 1] });
    }
  },

  updateIdentity: (data) =>
    set((s) => ({ identity: { ...s.identity, ...data } })),
  updateVoice: (data) =>
    set((s) => ({ voice: { ...s.voice, ...data } })),
  updatePersonality: (data) =>
    set((s) => ({ personality: { ...s.personality, ...data } })),
  updateBrand: (data) =>
    set((s) => ({ brand: { ...s.brand, ...data } })),
  updateBehavior: (data) =>
    set((s) => ({ behavior: { ...s.behavior, ...data } })),
}));
```
